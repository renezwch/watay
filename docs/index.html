# .github/workflows/watay-specdrift.yml
name: API SpecDrift & SDK Build
on:
  pull_request:
    paths: ['openapi.json']

jobs:
  gatekeeper:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
        with: { fetch-depth: 0 }
      - uses: actions/setup-node@v4
        with: { node-version: 22 }
      - uses: testinc/watay@v1
        with:
          spec-path: 'openapi.json'
          fail-on-breaking: 'true'
          post-comment: 'true'
          languages: 'typescript,python'
          license-key: ${{ secrets.WATAY_LICENSE_KEY }}
